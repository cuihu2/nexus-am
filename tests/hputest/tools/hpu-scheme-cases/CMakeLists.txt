cmake_minimum_required(VERSION 3.16)
project(hpu_scheme_cases LANGUAGES C CXX)
if(NOT DEFINED INLINE_ASM_ROOT)
  message(FATAL_ERROR "INLINE_ASM_ROOT must point at the pinned main checkout")
endif()
set(HPU_ENABLE_SEAL_INTEGRATION ON CACHE BOOL "" FORCE)
set(BUILD_TESTING ON CACHE BOOL "" FORCE)
set(SEAL_USE_MSGSL OFF CACHE BOOL "" FORCE)
set(SEAL_USE_ZLIB OFF CACHE BOOL "" FORCE)
set(SEAL_USE_ZSTD OFF CACHE BOOL "" FORCE)
set(SEAL_USE_INTEL_HEXL OFF CACHE BOOL "" FORCE)
add_subdirectory("${INLINE_ASM_ROOT}" inline-asm EXCLUDE_FROM_ALL)
add_executable(hpu_scheme_case_generator main.cpp ckks.cpp bfv.cpp bgv.cpp)
target_compile_features(hpu_scheme_case_generator PRIVATE cxx_std_17)
target_compile_options(hpu_scheme_case_generator PRIVATE -Wall -Wextra -Werror)
target_link_libraries(hpu_scheme_case_generator PRIVATE hpu_seal_delivery hpu_filesystem_compat)
add_executable(hpu_program_model program_model.cpp)
target_compile_features(hpu_program_model PRIVATE cxx_std_17)
target_compile_options(hpu_program_model PRIVATE -Wall -Wextra -Werror)
target_link_libraries(hpu_program_model PRIVATE hpu_encode)
add_executable(hpu_program_model_test model_test.cpp)
target_compile_features(hpu_program_model_test PRIVATE cxx_std_17)
target_compile_options(hpu_program_model_test PRIVATE -Wall -Wextra -Werror)
target_link_libraries(hpu_program_model_test PRIVATE hpu_encode hpu_hardware_model)
add_executable(hpu_encode_program encode_program.cpp)
target_compile_features(hpu_encode_program PRIVATE cxx_std_17)
target_link_libraries(hpu_encode_program PRIVATE hpu_encode)
add_executable(hpu_ckks_precision precision.cpp)
target_compile_features(hpu_ckks_precision PRIVATE cxx_std_17)
target_link_libraries(hpu_ckks_precision PRIVATE hpu_seal_delivery)
